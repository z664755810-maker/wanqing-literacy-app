/// 晚晴识字 · 内容数据（拼音 / 汉字 / 阅读）
///
/// 设计原则：
/// - 拼音：每个声母/韵母 clickable，进入详情页做完整发音讲解。
/// - 汉字：系统课程教单字；文化分类以“词条”形式呈现，内容贴近母亲生活。
/// - 笔画：内置正楷笔顺名称，复杂字给出完整顺序。
/// - 阅读：成语故事、寓言故事 + 生活短文，每个字上方显示拼音，点字发音。

/// 单字信息
class CharInfo {
  final String py; // 拼音（带调号）
  final String mean; // 释义
  final String ex; // 例词（用顿号分隔）
  final String sent; // 例句
  final List<String> strokes; // 笔顺名称，例如 ['横', '竖', '撇', '捺']
  const CharInfo(this.py, this.mean, this.ex, this.sent, [this.strokes = const []]);
}

/// 声母
class Shengmu {
  final String b; // 显示字母，如 b / zh
  final String name; // 名称，如“玻”
  final String py; // 读音，如 bō
  final String word; // 例字，如“爸爸”
  final String how; // 发音要领
  final String? warn; // 特别提醒
  const Shengmu(this.b, this.name, this.py, this.word, this.how, [this.warn]);
}

/// 韵母
class Yunmu {
  final String b; // 如 a / ang / ün
  final String name; // 名称，如“啊”
  final String py; // 读音，如 ā
  final String word; // 例字，如“妈妈”
  final String how; // 发音要领
  final List<String> tones; // 四声+轻声例词（只保留词，不保留拼音符号，避免TTS读错）
  const Yunmu(this.b, this.name, this.py, this.word, this.how, this.tones);
}

/// 词条（词组/节气名/节日名/成语/短语等）
class TermInfo {
  final String term; // 如“立春”
  final String py; // 拼音
  final String tag; // 分类标签
  final String meaning; // 释义
  final String sentence; // 例句
  final List<String> chars; // 组成单字
  const TermInfo(this.term, this.py, this.tag, this.meaning, this.sentence, this.chars);
}

/// 单字课程分类
class Cat {
  final String cat;
  final List<String> list;
  const Cat(this.cat, this.list);
}

/// 词条分类
class TermCat {
  final String cat;
  final List<String> list; // term 列表
  const TermCat(this.cat, this.list);
}

/// 阅读故事
class Story {
  final String title;
  final String tag;
  final String titlePy;
  final List<String> paragraphs; // 每段原文
  const Story(this.title, this.tag, this.titlePy, this.paragraphs);
}
